   211→
   212→/**
   213→ * One-shot highlight for a deep-linked section: a faint espresso wash plus a
   214→ * hairline espresso bar that fade away (tw-animate-css exit animation, held at
   215→ * the end state). The global prefers-reduced-motion override collapses the
   216→ * animation to its end state instantly, so nothing flashes there.
   217→ */
   218→function Flash() {
   219→  return (
   220→    <>
   221→      <span
   222→        aria-hidden
   223→        className="pointer-events-none absolute inset-0 bg-espresso/5 animate-out fade-out fill-mode-forwards duration-[2000ms] ease-in"
   224→      />
   225→      <span
   226→        aria-hidden
   227→        className="pointer-events-none absolute inset-y-0 left-0 w-[2px] bg-espresso animate-out fade-out fill-mode-forwards duration-[2000ms] ease-in"
   228→      />
   229→    </>
   230→  )
   231→}

   234→  id,

   236→  eyebrow,
   237→  title,
   238→  lede,

   247→  lede?: ReactNode
