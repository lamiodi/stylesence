   204→/* ------------------------------------------------------------------ *
   205→ * Back-in-stock waitlist — shown when the selected variant is sold out.
   206→ * Remounts via `key={variant.id}` so switching variants resets the form
   207→ * (the email prefill survives — it re-derives from the customer query).

   209→function StockWaitlistForm({
   210→  slug,
   211→  variantId,
   212→  color,
   213→  size,
   214→}: {
   215→  slug: string
   216→  variantId: string
   217→  color: string
   218→  size: string
   219→}) {
   220→  const { data: customer } = useCustomer()
   221→  const [email, setEmail] = useState('')
   222→  const [touched, setTouched] = useState(false)
   223→  const [busy, setBusy] = useState(false)
   224→  const [error, setError] = useState<string | null>(null)
   225→  const [done, setDone] = useState<{ email: string; already: boolean } | null>(null)
   226→
   227→  // Signed-in customers get their email prefilled — but only while the field
   228→  // is empty/untouched; anything typed always wins (checkout's prefill pattern).
   229→  const displayed = touched || email !== '' ? email : (customer?.email ?? '')
   230→
   231→  const submit = async (e: React.FormEvent) => {
   232→    e.preventDefault()
   233→    const value = displayed.trim()
   234→    if (!value) {
   235→      setError('Please enter your email address.')
   236→      return
   237→    }
   238→    setBusy(true)
   239→    setError(null)
   240→    try {
   241→      const res = await fetch(`/api/products/${slug}/stock-alerts`, {
   242→        method: 'POST',
   243→        headers: { 'Content-Type': 'application/json' },
   244→        body: JSON.stringify({ variantId, email: value }),
   245→      })
   246→      const data = (await res.json().catch(() => ({}))) as { ok?: boolean; alreadyWaiting?: boolean; error?: string }
   247→      if (!res.ok) throw new Error(data.error ?? 'Could not join the waitlist')
   248→      setDone({ email: value, already: data.alreadyWaiting === true })
   249→    } catch (err) {
   250→      setError(err instanceof Error ? err.message : 'Could not join the waitlist')
   251→    } finally {
   252→      setBusy(false)
   253→    }
   254→  }
   255→
   256→  if (done) {
   257→    return (
   258→      <div className="border border-line bg-secondary/50 p-4" role="status">
   259→        <div className="flex items-start gap-3">
   260→          <Check className="mt-0.5 h-4 w-4 shrink-0 text-espresso" strokeWidth={1.5} aria-hidden />
   261→          <div className="min-w-0">
   262→            <p className="eyebrow">On the list</p>
   263→            <p className="mt-1.5 text-sm leading-relaxed">
   264→              {done.already
   265→                ? `You’re already on the list — we’ll write when ${color} · ${size} returns.`
   266→                : `You’re on the list — we’ll write when ${color} · ${size} returns.`}
   267→            </p>
   268→            <p className="mt-1 truncate font-mono text-[0.68rem] text-muted-foreground">{done.email}</p>
   269→          </div>
   270→        </div>
   271→      </div>
   272→    )
   273→  }
   274→
   275→  return (
   276→    <form onSubmit={submit} noValidate className="border border-line bg-secondary/50 p-4">
   277→      <p className="eyebrow">Back in stock</p>
   278→      <p className="mt-1.5 text-[0.72rem] leading-relaxed text-muted-foreground">
   279→        We’ll write the moment {color} · {size} returns.
   280→      </p>
   281→      <div className="mt-3 flex items-center gap-2">
   282→        <div className="flex h-11 min-w-0 flex-1 items-center gap-2 border border-line-strong px-3 transition-colors focus-within:border-foreground">
   283→          <Mail className="h-3.5 w-3.5 shrink-0 text-muted-foreground" strokeWidth={1.5} aria-hidden />
   284→          <input
   285→            type="email"
   286→            autoComplete="email"
   287→            value={displayed}
   288→            onChange={(e) => {
   289→              setTouched(true)
   290→              setEmail(e.target.value)
   291→            }}
   292→            disabled={busy}
   293→            aria-label="Email for back-in-stock notification"
   294→            placeholder="you@example.com"
   295→            className="w-full bg-transparent text-sm placeholder:text-muted-foreground/60 focus:outline-none"
   296→          />
   297→        </div>
   298→        <button
   299→          type="submit"
   300→          disabled={busy}
   301→          className="h-11 shrink-0 border border-line-strong px-4 text-[0.62rem] font-medium uppercase tracking-[0.18em] text-muted-foreground transition-colors hover:border-foreground hover:text-foreground disabled:opacity-50"
   302→        >
   303→          {busy ? '…' : 'Notify me'}
   304→        </button>
   305→      </div>
   306→      {error ? (
   307→        <p role="alert" className="mt-2 text-[0.72rem] font-medium text-destructive">
   308→          {error}
   309→        </p>
   310→      ) : null}
   311→    </form>
   312→  )
   313→}
