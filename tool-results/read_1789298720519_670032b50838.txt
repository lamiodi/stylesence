   444→function ProfileForm({ customer }: { customer: CustomerView }) {
   445→  const update = useUpdateCustomerProfile()
   446→  const [name, setName] = useState(() => customer.name)
   447→  const [phone, setPhone] = useState(() => customer.phone ?? '')
   448→  const [address, setAddress] = useState(() => customer.defaultAddress ?? '')
   449→  const [city, setCity] = useState(() => customer.defaultCity ?? '')
   450→  const [state, setState] = useState(() =>
   451→    customer.defaultState && NG_STATES.includes(customer.defaultState) ? customer.defaultState : '',
   452→  )
   453→  const [errors, setErrors] = useState<Record<string, string>>({})
   454→
   455→  const validate = (): boolean => {
   456→    const e: Record<string, string> = {}
   457→    if (name.trim().length < 2) e.name = 'Your name is required.'
   458→    if (phone.trim().length > 0 && phone.trim().length < 7) e.phone = 'A reachable phone number is required.'
   459→    if (address.trim().length > 0 && address.trim().length < 5) e.address = 'Your street address is required.'
   460→    if (city.trim().length > 0 && city.trim().length < 2) e.city = 'Your city is required.'
   461→    setErrors(e)
   462→    return Object.keys(e).length === 0
   463→  }
   464→
   465→  const onSubmit = (e: FormEvent<HTMLFormElement>) => {
   466→    e.preventDefault()
   467→    if (!validate()) return
   468→    // Empty strings clear server-side (stored as null); absent fields are untouched.
   469→    update.mutate(
   470→      {
   471→        name: name.trim(),
   472→        phone: phone.trim(),
   473→        defaultAddress: address.trim(),
   474→        defaultCity: city.trim(),
   475→        defaultState: state,
   476→      },
   477→      {
   478→        onSuccess: (saved) => {
   479→          setName(saved.name)
   480→          setPhone(saved.phone ?? '')
   481→          setAddress(saved.defaultAddress ?? '')
   482→          setCity(saved.defaultCity ?? '')
   483→          setState(saved.defaultState ?? '')
   484→        },
   485→      },
   486→    )
   487→  }
   488→
   489→  const err = (k: string) =>
   490→    errors[k] ? (
   491→      <p className="text-[0.7rem] text-destructive" role="alert">
   492→        {errors[k]}
   493→      </p>
   494→    ) : null
   495→
   496→  return (
   497→    <section aria-label="Profile and default shipping details" className="border border-line bg-card p-6 sm:p-8">
   498→      <p className="eyebrow">Saved details</p>
   499→      <h2 className="mt-2 font-display text-2xl font-light tracking-tight">Profile & delivery</h2>
   500→      <p className="mt-2 text-[0.8rem] leading-relaxed text-muted-foreground">
   501→        These prefill checkout — set them once, shop faster after.
   502→      </p>
   503→
   504→      <form onSubmit={onSubmit} noValidate className="mt-6 space-y-4">
   505→        <div className="space-y-1.5">
   506→          <Label htmlFor="pf-name" className="eyebrow">Name</Label>
   507→          <Input
   508→            id="pf-name"
   509→            type="text"
   510→            autoComplete="name"
   511→            value={name}
   512→            onChange={(e) => setName(e.target.value)}
   513→            className={cn(fieldCls, errors.name && 'border-destructive')}
   514→            aria-invalid={errors.name ? true : undefined}
   515→          />
   516→          {err('name')}
   517→        </div>
   518→        <div className="space-y-1.5">
   519→          <Label htmlFor="pf-phone" className="eyebrow">Phone</Label>
   520→          <Input
   521→            id="pf-phone"
   522→            type="tel"
   523→            autoComplete="tel"
   524→            value={phone}
   525→            onChange={(e) => setPhone(e.target.value)}
   526→            placeholder="+234 801 234 5678"
   527→            className={cn(fieldCls, errors.phone && 'border-destructive')}
   528→            aria-invalid={errors.phone ? true : undefined}
   529→          />
   530→          {err('phone')}
   531→        </div>
   532→        <div className="space-y-1.5">
   533→          <Label htmlFor="pf-address" className="eyebrow">Default address</Label>
   534→          <Input
   535→            id="pf-address"
   536→            type="text"
   537→            autoComplete="street-address"
   538→            value={address}
   539→            onChange={(e) => setAddress(e.target.value)}
   540→            placeholder="14A Awolowo Road, Ikoyi"
   541→            className={cn(fieldCls, errors.address && 'border-destructive')}
   542→            aria-invalid={errors.address ? true : undefined}
   543→          />
   544→          {err('address')}
   545→        </div>
   546→        <div className="grid gap-4 sm:grid-cols-2">
   547→          <div className="space-y-1.5">
   548→            <Label htmlFor="pf-city" className="eyebrow">City</Label>
   549→            <Input
   550→              id="pf-city"
   551→              type="text"
   552→              autoComplete="address-level2"
   553→              value={city}
   554→              onChange={(e) => setCity(e.target.value)}
   555→              placeholder="Lagos"
   556→              className={cn(fieldCls, errors.city && 'border-destructive')}
   557→              aria-invalid={errors.city ? true : undefined}
   558→            />
   559→            {err('city')}
   560→          </div>
   561→          <div className="space-y-1.5">
   562→            <Label htmlFor="pf-state" className="eyebrow">State</Label>
   563→            <select
   564→              id="pf-state"
   565→              value={state}
   566→              onChange={(e) => setState(e.target.value)}
   567→              className={cn(

   647→async function fetchCustomerOrders(): Promise<CustomerOrderSummary[]> {
   648→  const res = await fetch('/api/customer/orders')
   649→  const body = (await res.json().catch(() => ({}))) as { orders?: CustomerOrderSummary[]; error?: string }
   650→  if (!res.ok || !body.orders) throw new Error(body.error ?? 'Order history is unavailable right now.')
   651→  return body.orders
   652→}
