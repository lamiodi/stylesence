    12→  const salt = crypto.randomBytes(16).toString('hex')
    13→  const hash = crypto.scryptSync(password, salt, 64).toString('hex')

    17→// Deterministic PRNG so seeds are stable
    18→function mulberry32(a: number) {
    19→  return function () {
    20→    let t = (a += 0x6d2b79f5)
    21→    t = Math.imul(t ^ (t >>> 15), t | 1)
    22→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61)
    23→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
    24→  }
