    49→/* ——— recent-lookup history (localStorage, guarded) ——— */
    50→
    51→function readHistory(): string[] {
    52→  if (typeof window === 'undefined') return []
    53→  try {
    54→    const raw = window.localStorage.getItem(HISTORY_KEY)
    55→    if (!raw) return []
    56→    const parsed: unknown = JSON.parse(raw)
    57→    if (!Array.isArray(parsed)) return []
    58→    return parsed
    59→      .filter((v): v is string => typeof v === 'string' && ORDER_PATTERN.test(v))
    60→      .slice(0, HISTORY_MAX)
    61→  } catch {
    62→    return []
    63→  }
    64→}
    65→
    66→/** Remember a successful lookup — most recent first, distinct, capped. Returns the next list. */
    67→function rememberLookup(orderNumber: string): string[] {
    68→  const next = [orderNumber, ...readHistory().filter((n) => n !== orderNumber)].slice(0, HISTORY_MAX)
    69→  try {
    70→    window.localStorage.setItem(HISTORY_KEY, JSON.stringify(next))
    71→  } catch {
    72→    // Storage unavailable (private mode / blocked) — this session keeps the list in state only.
    73→  }
    74→  return next
    75→}
