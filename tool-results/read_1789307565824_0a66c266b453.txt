   203→/* ------------------------------------------------------------------ *
   204→ * Back-in-stock waitlist — shown when the selected variant is sold out.
   205→ * Remounts via `key={variant.id}` so switching variants resets the form
   206→ * (the email prefill survives — it re-derives from the customer query).

   208→function StockWaitlistForm({
   209→  slug,
   210→  variantId,
   211→  color,
   212→  size,
   213→}: {
   214→  slug: string
   215→  variantId: string
   216→  color: string
   217→  size: string
   218→}) {
   219→  const { data: customer } = useCustomer()
   220→  const [email, setEmail] = useState('')
   221→  const [touched, setTouched] = useState(false)
   222→  const [busy, setBusy] = useState(false)
   223→  const [error, setError] = useState<string | null>(null)
   224→  const [done, setDone] = useState<{ email: string; already: boolean } | null>(null)
   225→
   226→  // Signed-in customers get their email prefilled — but only while the field
   227→  // is empty/untouched; anything typed always wins (checkout's prefill pattern).
   228→  const displayed = touched || email !== '' ? email : (customer?.email ?? '')
   229→
   230→  const submit = async (e: React.FormEvent) => {
   231→    e.preventDefault()
   232→    const value = displayed.trim()
   233→    if (!value) {
   234→      setError('Please enter your email address.')
   235→      return
   236→    }
   237→    setBusy(true)
   238→    setError(null)
   239→    try {
   240→      const res = await fetch(`/api/products/${slug}/stock-alerts`, {
   241→        method: 'POST',
   242→        headers: { 'Content-Type': 'application/json' },
   243→        body: JSON.stringify({ variantId, email: value }),
   244→      })
   245→      const data = (await res.json().catch(() => ({}))) as { ok?: boolean; alreadyWaiting?: boolean; error?: string }
   246→      if (!res.ok) throw new Error(data.error ?? 'Could not join the waitlist')
   247→      setDone({ email: value, already: data.alreadyWaiting === true })
   248→    } catch (err) {
   249→      setError(err instanceof Error ? err.message : 'Could not join the waitlist')
   250→    } finally {
   251→      setBusy(false)
   252→    }
   253→  }
   254→
   255→  if (done) {
   256→    return (
   257→      <div className="border border-line bg-secondary/50 p-4" role="status">
   258→        <div className="flex items-start gap-3">
   259→          <Check className="mt-0.5 h-4 w-4 shrink-0 text-espresso" strokeWidth={1.5} aria-hidden />
   260→          <div className="min-w-0">
   261→            <p className="eyebrow">On the list</p>
   262→            <p className="mt-1.5 text-sm leading-relaxed">
   263→              {done.already
   264→                ? `You’re already on the list — we’ll write when ${color} · ${size} returns.`
   265→                : `You’re on the list — we’ll write when ${color} · ${size} returns.`}
   266→            </p>
   267→            <p className="mt-1 truncate font-mono text-[0.68rem] text-muted-foreground">{done.email}</p>
   268→          </div>
   269→        </div>
   270→      </div>
   271→    )
   272→  }
   273→
   274→  return (
   275→    <form onSubmit={submit} noValidate className="border border-line bg-secondary/50 p-4">
   276→      <p className="eyebrow">Back in stock</p>
   277→      <p className="mt-1.5 text-[0.72rem] leading-relaxed text-muted-foreground">
   278→        We’ll write the moment {color} · {size} returns.
   279→      </p>
   280→      <div className="mt-3 flex items-center gap-2">
   281→        <div className="flex h-11 min-w-0 flex-1 items-center gap-2 border border-line-strong px-3 transition-colors focus-within:border-foreground">
   282→          <Mail className="h-3.5 w-3.5 shrink-0 text-muted-foreground" strokeWidth={1.5} aria-hidden />
   283→          <input
   284→            type="email"
   285→            autoComplete="email"
   286→            value={displayed}
   287→            onChange={(e) => {
   288→              setTouched(true)
   289→              setEmail(e.target.value)
   290→            }}
   291→            disabled={busy}
   292→            aria-label="Email for back-in-stock notification"
   293→            placeholder="you@example.com"
   294→            className="w-full bg-transparent text-sm placeholder:text-muted-foreground/60 focus:outline-none"
   295→          />
   296→        </div>
   297→        <button
   298→          type="submit"
   299→          disabled={busy}
   300→          className="h-11 shrink-0 border border-line-strong px-4 text-[0.62rem] font-medium uppercase tracking-[0.18em] text-muted-foreground transition-colors hover:border-foreground hover:text-foreground disabled:opacity-50"
   301→        >
   302→          {busy ? '…' : 'Notify me'}
   303→        </button>
   304→      </div>
   305→      {error ? (
   306→        <p role="alert" className="mt-2 text-[0.72rem] font-medium text-destructive">
   307→          {error}
   308→        </p>
   309→      ) : null}
   310→    </form>
   311→  )
   312→}

   378→    product.variants.length === 1 || product.variants.every((v) => v.size === 'One Size')
   379→      ? product.variants[0]?.size ?? null

   389→  const mounted = useMounted()
   390→

   397→    const io = new IntersectionObserver(
   398→      ([entry]) => {
   399→        // only when the actions left the viewport upwards (scrolled past, not approaching)
   400→        setShowStickyBar(!entry.isIntersecting && entry.boundingClientRect.top < 0)

   402→      { threshold: 0 },
