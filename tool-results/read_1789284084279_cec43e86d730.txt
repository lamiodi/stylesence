    89→function WriteReviewDialog({ slug, product }: { slug: string; product: string }) {
    90→  const [rating, setRating] = useState(5)
    91→  const [author, setAuthor] = useState('')
    92→  const [email, setEmail] = useState('')
    93→  const [title, setTitle] = useState('')
    94→  const [body, setBody] = useState('')
    95→  const [busy, setBusy] = useState(false)
    96→  const [open, setOpen] = useState(false)
    97→
    98→  const submit = async (e: React.FormEvent) => {
    99→    e.preventDefault()
   100→    if (author.trim().length < 2) return toast.error('Please add your name.')
   101→    if (body.trim().length < 10) return toast.error('Please write at least a sentence or two.')
   102→    setBusy(true)
   103→    try {
   104→      const res = await fetch(`/api/products/${slug}/reviews`, {
   105→        method: 'POST',
   106→        headers: { 'Content-Type': 'application/json' },
   107→        body: JSON.stringify({
   108→          author: author.trim(),
   109→          email: email.trim() || undefined,
   110→          rating,
   111→          title: title.trim() || undefined,
   112→          body: body.trim(),
   113→        }),
   114→      })
   115→      const data = await res.json()
   116→      if (!res.ok) throw new Error(data.error ?? 'Could not submit review')
   117→      toast.success('Thank you — your review is awaiting moderation.')
   118→      setOpen(false)
   119→      setAuthor('')
   120→      setEmail('')
   121→      setTitle('')
   122→      setBody('')
   123→      setRating(5)
   124→    } catch (err) {
   125→      toast.error(err instanceof Error ? err.message : 'Could not submit review')
   126→    } finally {
   127→      setBusy(false)
   128→    }
   129→  }
   130→
   131→  return (
   132→    <Dialog open={open} onOpenChange={setOpen}>
   133→      <DialogTrigger asChild>
   134→        <Button
   135→          variant="outline"
   136→          className="h-11 border-line-strong px-6 uppercase tracking-[0.18em] text-[0.62rem] hover:border-foreground"
   137→        >
   138→          Write a review
   139→        </Button>
   140→      </DialogTrigger>
   141→      <DialogContent className="max-w-lg rounded-[--radius] border-line bg-background">
   142→        <DialogHeader>
   143→          <DialogTitle className="font-display text-2xl font-light">{product}</DialogTitle>
   144→          <DialogDescription>Your honest impressions — on the record.</DialogDescription>
   145→        </DialogHeader>
   146→        <form onSubmit={submit} className="space-y-4">
   147→          <div>
   148→            <Label className="eyebrow">Your rating</Label>
   149→            <div className="mt-2 flex items-center gap-1.5">
   150→              {[1, 2, 3, 4, 5].map((star) => (
   151→                <button
   152→                  key={star}
   153→                  type="button"
   154→                  onClick={() => setRating(star)}
   155→                  aria-label={`${star} star${star > 1 ? 's' : ''}`}
   156→                  aria-pressed={rating === star}
   157→                  className={cn(
   158→                    'flex h-9 w-9 items-center justify-center border transition-colors focus-visible:outline-2 focus-visible:outline-ring',
   159→                    rating >= star ? 'border-espresso' : 'border-line-strong',
   160→                  )}
   161→                >
   162→                  <span className={cn('text-sm', rating >= star ? 'text-espresso' : 'text-muted-foreground/50')}>
   163→                    ★
   164→                  </span>
   165→                </button>
   166→              ))}
   167→              <span className="ml-2 font-mono text-sm tabular-nums text-muted-foreground">{rating}.0</span>
   168→            </div>
   169→          </div>
   170→          <div className="grid gap-4 sm:grid-cols-2">
   171→            <div className="space-y-1.5">
   172→              <Label htmlFor="rv-name" className="eyebrow">Name *</Label>
   173→              <Input id="rv-name" value={author} onChange={(e) => setAuthor(e.target.value)} placeholder="Adaeze O." className="h-10 border-line-strong" />
   174→            </div>
   175→            <div className="space-y-1.5">
   176→              <Label htmlFor="rv-email" className="eyebrow">Email (optional)</Label>
   177→              <Input id="rv-email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} placeholder="you@example.com" className="h-10 border-line-strong" />
   178→            </div>
   179→          </div>
   180→          <div className="space-y-1.5">
   181→            <Label htmlFor="rv-title" className="eyebrow">Title</Label>
   182→            <Input id="rv-title" value={title} onChange={(e) => setTitle(e.target.value)} placeholder="It moves like water" className="h-10 border-line-strong" />
   183→          </div>
   184→          <div className="space-y-1.5">
   185→            <Label htmlFor="rv-body" className="eyebrow">Your review *</Label>
   186→            <Textarea id="rv-body" value={body} onChange={(e) => setBody(e.target.value)} rows={4} placeholder="Fit, feel, fabric — tell it as it is." className="border-line-strong" />

   194→            </Button>
   195→          </div>
   196→        </form>
   197→      </DialogContent>

   202→export function ProductDetailPage({ slug }: { slug: string }) {
   203→  const { data, isLoading, isError } = useQuery({
   204→    queryKey: ['product', slug],
   205→    queryFn: async () => {
   206→      const res = await fetch(`/api/products/${slug}`)

   208→      return (await res.json()) as { product: ProductDetail }
   209→    },
   210→    retry: false,

   212→
   213→  const product = data?.product
   214→
   215→  useEffect(() => {

   218→
   219→  if (isLoading) {

   261→function ProductInner({ product }: { product: ProductDetail }) {
   262→  const [color, setColor] = useState<string | null>(product.variants[0]?.color ?? null)
   263→  const [size, setSize] = useState<string | null>(
   264→    product.variants.length === 1 ? product.variants[0]?.size ?? null : null,
   265→  )
   266→  const [qty, setQty] = useState(1)
   267→  const [imgIndex, setImgIndex] = useState(0)
   268→  const addToCart = useAddToCart()
   269→  const toggleWish = useWishlist((s) => s.toggle)
   270→  const hasWish = useWishlist((s) => s.has)
   271→  const pushRecent = useRecentlyViewed((s) => s.push)
   272→  const mounted = useMounted()
   273→
   274→  // sticky mobile buy bar — appears once the main actions scroll out above
   275→  const actionsRef = useRef<HTMLDivElement>(null)
   276→  const [showStickyBar, setShowStickyBar] = useState(false)
   277→  useEffect(() => {
   278→    const el = actionsRef.current
   279→    if (!el || typeof IntersectionObserver === 'undefined') return
   280→    const io = new IntersectionObserver(
   281→      ([entry]) => {
   282→        // only when the actions left the viewport upwards (scrolled past, not approaching)
   283→        setShowStickyBar(!entry.isIntersecting && entry.boundingClientRect.top < 0)
   284→      },
   285→      { threshold: 0 },
   286→    )
   287→    io.observe(el)
   288→    return () => io.disconnect()
   289→  }, [])
   290→
   291→  // record the visit once per product (external store — safe inside an effect)
   292→  useEffect(() => {
   293→    pushRecent({
   294→      slug: product.slug,
   295→      name: product.name,
   296→      price: product.price,
   297→      primaryImage: product.images[0]?.url ?? null,
   298→      secondaryImage: product.images[1]?.url ?? null,
   299→    })
   300→  }, [product, pushRecent])
   301→
   302→  const colors = useMemo(() => {
   303→    const seen = new Map<string, string>()
   304→    for (const v of product.variants) if (!seen.has(v.color)) seen.set(v.color, v.colorHex)
   305→    return [...seen.entries()].map(([name, hex]) => ({ name, hex }))
   306→  }, [product])
   307→
   308→  const sizesForColor = useMemo(
   309→    () => (color ? product.variants.filter((v) => v.color === color) : []),
   310→    [product, color],
   311→  )
   312→
   313→  const selectedVariant = useMemo(
   314→    () => product.variants.find((v) => v.color === color && v.size === size) ?? null,
   315→    [product, color, size],
   316→  )
   317→
   318→  const wished = mounted && hasWish(product.slug)
   319→  const oneSize = product.variants.every((v) => v.size === 'One Size')
   320→  const stockNote =
   321→    selectedVariant && selectedVariant.stock === 0
   322→      ? 'Sold out — join the waitlist below'
   323→      : selectedVariant && selectedVariant.stock <= 3
   324→        ? `Only ${selectedVariant.stock} left in this colour & size`
   325→        : null
   326→
   327→  const distribution = [5, 4, 3, 2, 1].map((star) => ({
   328→    star,
   329→    count: product.reviews.filter((r) => r.rating === star).length,
   330→  }))
   331→  const maxCount = Math.max(1, ...distribution.map((d) => d.count))
   332→
   333→  return (
   334→    <div className="container-site py-8 sm:py-12">
   335→      {/* breadcrumb */}
   336→      <nav aria-label="Breadcrumb" className="eyebrow !text-[0.58rem]">
   337→        <Link to="/shop" className="hover:text-foreground">Shop</Link>
   338→        {product.category ? (
   339→          <>
   340→            <ChevronRight className="mx-1.5 inline h-2.5 w-2.5" aria-hidden />
   341→            <Link to={`/shop?category=${product.category.slug}`} className="hover:text-foreground">
   342→              {product.category.name}
   343→            </Link>
   344→          </>
   345→        ) : null}
   346→        <ChevronRight className="mx-1.5 inline h-2.5 w-2.5" aria-hidden />
   347→        <span className="text-foreground">{product.name}</span>
   348→      </nav>
   349→
   350→      <div className="mt-6 grid gap-10 lg:grid-cols-[1.15fr_1fr] lg:gap-16">
   351→        {/* ————— gallery ————— */}
   352→        <div>
   353→          <div className="relative">
   354→            <ProductImage
   355→              key={product.images[imgIndex]?.url ?? 'none'}
   356→              src={product.images[imgIndex]?.url}
   357→              alt={product.images[imgIndex]?.alt ?? product.name}
   358→              label={product.name}
   359→              ratio="aspect-[3/4]"
   360→              eager
   361→              className="w-full"
   362→            />
   363→            {product.compareAtPrice && product.compareAtPrice > product.price ? (
   364→              <span className="eyebrow absolute left-4 top-4 bg-espresso px-2.5 py-1 !text-[0.55rem] text-background">
   365→                Archive price
   366→              </span>
   367→            ) : null}
   368→          </div>
   369→          {product.images.length > 1 ? (
   370→            <div className="mt-3 flex gap-3">
   371→              {product.images.map((img, i) => (
   372→                <button
   373→                  key={img.url}
   374→                  type="button"
   375→                  onClick={() => setImgIndex(i)}
   376→                  aria-label={`View image ${i + 1} of ${product.images.length}`}
   377→                  aria-pressed={imgIndex === i}
   378→                  className={cn(
   379→                    'w-20 border transition-colors focus-visible:outline-2 focus-visible:outline-ring',
   380→                    imgIndex === i ? 'border-foreground' : 'border-transparent hover:border-line-strong',
   381→                  )}
   382→                >
   383→                  <ProductImage src={img.url} alt={img.alt ?? product.name} label="" ratio="aspect-[3/4]" />
   384→                </button>
   385→              ))}
   386→            </div>
   387→          ) : null}
   388→        </div>
   389→
   390→        {/* ————— info ————— */}
   391→        <div className="lg:pt-2">
   392→          {product.category ? <p className="eyebrow">{product.category.name}</p> : null}
   393→          <h1 className="mt-2 font-display text-3xl font-light leading-tight tracking-tight text-balance sm:text-4xl">
   394→            {product.name}
   395→          </h1>
